14inline std::vector<double> jacobi_eigvals(const Arr& A) {
15 auto n = A.rows();
16 Arr a = A;
17 for (size_t sweep = 0; sweep < 100; ++sweep) {
18 double off = 0.0;
19 for (size_t p = 0; p < n; ++p)
20 for (size_t q = p + 1; q < n; ++q) off += a(p, q) * a(p, q);
21 if (off <= 0.0) break;
22 for (size_t p = 0; p < n; ++p) {
23 for (size_t q = p + 1; q < n; ++q) {
24 double apq = a(p, q);
25 if (apq == 0.0) continue;
26 double app = a(p, p);
27 double aqq = a(q, q);
28 double theta = (aqq - app) / (2.0 * apq);
29 double sgn = (theta >= 0.0) ? 1.0 : -1.0;
30 double tt = sgn / (std::abs(theta) + std::sqrt(theta * theta + 1.0));
31 double c = 1.0 / std::sqrt(tt * tt + 1.0);
32 double s = tt * c;
33 for (size_t r = 0; r < n; ++r) {
34 if (r != p && r != q) {
35 double arp = a(r, p);
36 double arq = a(r, q);
37 a(r, p) = c * arp - s * arq;
38 a(p, r) = a(r, p);
39 a(r, q) = s * arp + c * arq;
40 a(q, r) = a(r, q);
41 }
42 }
43 a(p, p) = c * c * app - 2.0 * s * c * apq + s * s * aqq;
44 a(q, q) = s * s * app + 2.0 * s * c * apq + c * c * aqq;
45 a(p, q) = 0.0;
46 a(q, p) = 0.0;
47 }
48 }
49 }
50 std::vector<double> ev(n);
51 for (size_t i = 0; i < n; ++i) ev[i] = a(i, i);
52 std::sort(ev.begin(), ev.end());
53 return ev;
54}

98 inline DD dd_div(DD a, DD b) {
99 double q1 = a.hi / b.hi;
100 DD r = dd_sub(a, dd_mul(b, {q1, 0.0}));
101 double q2 = r.hi / b.hi;
102 r = dd_sub(r, dd_mul(b, {q2, 0.0}));
103 double q3 = r.hi / b.hi;
104 DD q = two_sum(q1, q2);
105 q.lo += q3;
106 return quick_two_sum(q.hi, q.lo);

109 inline DD dd_sqrt(DD a) {
110 if (a.hi <= 0.0) return {0.0, 0.0};
111 double x = std::sqrt(a.hi);
112 DD diff = dd_sub(a, two_prod(x, x));
113 return quick_two_sum(x, diff.hi / (2.0 * x));

122inline double design_cond(const std::vector<Arr>& Sigma) {
123 using dd_detail::DD;
124 auto m = Sigma.size();
125 auto n2 = Sigma[0].size();
126 std::vector<DD> G(m * m);
127 double dmax = 0.0;
128 for (size_t p = 0; p < m; ++p) {
129 for (size_t q = p; q < m; ++q) {
130 DD s{0.0, 0.0};
131 for (size_t i = 0; i < n2; ++i)
132 s = dd_detail::dd_add(s, dd_detail::two_prod(Sigma[p](i), Sigma[q](i)));
133 G[p * m + q] = s;
134 G[q * m + p] = s;
135 if (p == q) dmax = std::max(dmax, std::abs(s.hi));
136 }
137 }
138 auto at = [&](size_t i, size_t j) -> DD& { return G[i * m + j]; };
139 const double thresh = 1e-31 * dmax;
140 for (size_t sweep = 0; sweep < 80; ++sweep) {
141 bool changed = false;
142 for (size_t p = 0; p < m; ++p) {
143 for (size_t q = p + 1; q < m; ++q) {
144 DD apq = at(p, q);
145 if (std::abs(apq.hi) <= thresh) continue;
146 changed = true;
147 DD app = at(p, p);
148 DD aqq = at(q, q);
149 DD theta = dd_detail::dd_div(dd_detail::dd_sub(aqq, app),
150 DD{2.0 * apq.hi, 2.0 * apq.lo});
151 DD abtheta = (theta.hi >= 0.0) ? theta : dd_detail::dd_neg(theta);
152 DD denom = dd_detail::dd_add(
153 abtheta, dd_detail::dd_sqrt(
154 dd_detail::dd_add(DD{1.0, 0.0}, dd_detail::dd_mul(theta, theta))));
155 DD tt = dd_detail::dd_div(DD{(theta.hi >= 0.0) ? 1.0 : -1.0, 0.0}, denom);
156 DD cc = dd_detail::dd_div(
157 DD{1.0, 0.0},
158 dd_detail::dd_sqrt(dd_detail::dd_add(DD{1.0, 0.0}, dd_detail::dd_mul(tt, tt))));
159 DD ss = dd_detail::dd_mul(tt, cc);
160 for (size_t r = 0; r < m; ++r) {
161 if (r == p || r == q) continue;
162 DD arp = at(r, p);
163 DD arq = at(r, q);
164 DD np
165 = dd_detail::dd_sub(dd_detail::dd_mul(cc, arp), dd_detail::dd_mul(ss, arq));
166 DD nq
167 = dd_detail::dd_add(dd_detail::dd_mul(ss, arp), dd_detail::dd_mul(cc, arq));
168 at(r, p) = np;
169 at(p, r) = np;
170 at(r, q) = nq;
171 at(q, r) = nq;
172 }
173 DD cc2 = dd_detail::dd_mul(cc, cc);
174 DD ss2 = dd_detail::dd_mul(ss, ss);
175 DD cs2 = dd_detail::dd_mul(DD{2.0, 0.0}, dd_detail::dd_mul(ss, cc));
176 at(p, p) = dd_detail::dd_add(
177 dd_detail::dd_sub(dd_detail::dd_mul(cc2, app), dd_detail::dd_mul(cs2, apq)),
178 dd_detail::dd_mul(ss2, aqq));
179 at(q, q) = dd_detail::dd_add(
180 dd_detail::dd_add(dd_detail::dd_mul(ss2, app), dd_detail::dd_mul(cs2, apq)),
181 dd_detail::dd_mul(cc2, aqq));
182 at(p, q) = DD{0.0, 0.0};
183 at(q, p) = DD{0.0, 0.0};
184 }
185 }
186 if (!changed) break;
187 }
188 DD lmin = at(0, 0);
189 DD lmax = at(0, 0);
190 for (size_t i = 1; i < m; ++i) {
191 if (at(i, i).hi < lmin.hi) lmin = at(i, i);
192 if (at(i, i).hi > lmax.hi) lmax = at(i, i);
193 }
194 DD cond = dd_detail::dd_sqrt(dd_detail::dd_div(lmax, lmin));
195 return cond.hi + cond.lo;
