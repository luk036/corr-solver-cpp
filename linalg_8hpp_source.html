138inline Arr matmul(const Arr& A, const Arr& B) {
139 assert(A.is_2d() && B.is_2d() && A.cols() == B.rows());
140 auto m = A.rows();
141 auto k = A.cols();
142 auto n = B.cols();
143 Arr out(m, n);
144 // i-k-j order: `B(t, :)` and `out(i, :)` are both contiguous, so the inner
145 // loop auto-vectorizes and B streams through cache. The i-j-t order this
146 // replaces walked B one column at a time (stride n), which is cache-hostile.
147 for (size_t i = 0; i < m; ++i) {
148 for (size_t t = 0; t < k; ++t) {
149 const auto a = A(i, t);
150 for (size_t j = 0; j < n; ++j) out(i, j) += a * B(t, j);
151 }
152 }
153 return out;
154}

157// Cholesky decomposition: A = L * L^T (A symmetric positive definite)
158// Returns lower-triangular L.

187inline Arr cholesky(const Arr& A) {
188 auto n = A.rows();
189 assert(A.is_2d() && A.cols() == n);
190 Arr L(n, n);
191 for (size_t j = 0; j < n; ++j) {
192 double s = 0.0;
193 for (size_t k = 0; k < j; ++k) s += L(j, k) * L(j, k);
194 L(j, j) = std::sqrt(A(j, j) - s);
195 for (size_t i = j + 1; i < n; ++i) {
196 s = 0.0;
197 for (size_t k = 0; k < j; ++k) s += L(i, k) * L(j, k);
198 L(i, j) = (A(i, j) - s) / L(j, j);
199 }
200 }
201 return L;
202}

230inline Arr inv(const Arr& A) {
231 auto n = A.rows();
232 auto L = cholesky(A);
233 // Solve L * Y = I (forward substitution)
234 Arr Y(n, n);
235 for (size_t j = 0; j < n; ++j) {
236 for (size_t i = 0; i < n; ++i) {
237 double s = (i == j) ? 1.0 : 0.0;
238 for (size_t k = 0; k < i; ++k) s -= L(i, k) * Y(k, j);
239 Y(i, j) = s / L(i, i);
240 }
241 }
242 // Solve L^T * X = Y (back substitution)
243 Arr X(n, n);
244 for (size_t j = 0; j < n; ++j) {
245 for (size_t i = n; i-- > 0;) {
246 double s = Y(i, j);
247 for (size_t k = i + 1; k < n; ++k) s -= L(k, i) * X(k, j);
248 X(i, j) = s / L(i, i);
249 }
250 }
251 return X;
252}

254// ---------------------------------------------------------------------------
255// Inverse of an upper-triangular matrix via back substitution.
256// `inv` above assumes a symmetric SPD matrix; applying it to the Cholesky
257// factor R (upper triangular, A = R^T R) silently returns diag(1/R_ii) instead
258// of R^{-1}. This routine inverts R properly.
259// ---------------------------------------------------------------------------
260
272inline Arr inv_upper_tri(const Arr& r) {
273 auto n = r.rows();
274 Arr x(n, n);
275 for (size_t j = 0; j < n; ++j) {
276 for (size_t i = j + 1; i-- > 0;) {
277 double s = (i == j) ? 1.0 : 0.0;
278 for (size_t k = i + 1; k <= j; ++k) s -= r(i, k) * x(k, j);
279 x(i, j) = s / r(i, i);
280 }
281 }
282 return x;
283}
