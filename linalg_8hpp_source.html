107inline Arr matmul(const Arr& A, const Arr& B) {
108 assert(A.is_2d() && B.is_2d() && A.cols() == B.rows());
109 auto m = A.rows();
110 auto k = A.cols();
111 auto n = B.cols();
112 Arr out(m, n);
113 for (size_t i = 0; i < m; ++i)
114 for (size_t j = 0; j < n; ++j) {
115 double s = 0.0;
116 for (size_t t = 0; t < k; ++t) s += A(i, t) * B(t, j);
117 out(i, j) = s;
118 }
119 return out;
120}

122// ---------------------------------------------------------------------------
123// Cholesky decomposition: A = L * L^T (A symmetric positive definite)
124// Returns lower-triangular L.
125// ---------------------------------------------------------------------------
126
153inline Arr cholesky(const Arr& A) {
154 auto n = A.rows();
155 assert(A.is_2d() && A.cols() == n);
156 Arr L(n, n);
157 for (size_t j = 0; j < n; ++j) {
158 double s = 0.0;
159 for (size_t k = 0; k < j; ++k) s += L(j, k) * L(j, k);
160 L(j, j) = std::sqrt(A(j, j) - s);
161 for (size_t i = j + 1; i < n; ++i) {
162 s = 0.0;
163 for (size_t k = 0; k < j; ++k) s += L(i, k) * L(j, k);
164 L(i, j) = (A(i, j) - s) / L(j, j);
165 }
166 }
167 return L;
168}

196inline Arr inv(const Arr& A) {
197 auto n = A.rows();
198 auto L = cholesky(A);
199 // Solve L * Y = I (forward substitution)
200 Arr Y(n, n);
201 for (size_t j = 0; j < n; ++j) {
202 for (size_t i = 0; i < n; ++i) {
203 double s = (i == j) ? 1.0 : 0.0;
204 for (size_t k = 0; k < i; ++k) s -= L(i, k) * Y(k, j);
205 Y(i, j) = s / L(i, i);
206 }
207 }
208 // Solve L^T * X = Y (back substitution)
209 Arr X(n, n);
210 for (size_t j = 0; j < n; ++j) {
211 for (size_t i = n; i-- > 0;) {
212 double s = Y(i, j);
213 for (size_t k = i + 1; k < n; ++k) s -= L(k, i) * X(k, j);
214 X(i, j) = s / L(i, i);
215 }
216 }
217 return X;
218}
