134inline Arr matmul(const Arr& A, const Arr& B) {
135 assert(A.is_2d() && B.is_2d() && A.cols() == B.rows());
136 auto m = A.rows();
137 auto k = A.cols();
138 auto n = B.cols();
139 Arr out(m, n);
140 // i-k-j order: `B(t, :)` and `out(i, :)` are both contiguous, so the inner
141 // loop auto-vectorizes and B streams through cache. The i-j-t order this
142 // replaces walked B one column at a time (stride n), which is cache-hostile.
143 for (size_t i = 0; i < m; ++i) {
144 for (size_t t = 0; t < k; ++t) {
145 const auto a = A(i, t);
146 for (size_t j = 0; j < n; ++j) out(i, j) += a * B(t, j);
147 }
148 }
149 return out;
150}

153// Cholesky decomposition: A = L * L^T (A symmetric positive definite)

183inline Arr cholesky(const Arr& A) {
184 auto n = A.rows();
185 assert(A.is_2d() && A.cols() == n);
186 Arr L(n, n);
187 for (size_t j = 0; j < n; ++j) {
188 double s = 0.0;
189 for (size_t k = 0; k < j; ++k) s += L(j, k) * L(j, k);
190 L(j, j) = std::sqrt(A(j, j) - s);
191 for (size_t i = j + 1; i < n; ++i) {
192 s = 0.0;
193 for (size_t k = 0; k < j; ++k) s += L(i, k) * L(j, k);
194 L(i, j) = (A(i, j) - s) / L(j, j);
195 }
196 }
197 return L;
198}

226inline Arr inv(const Arr& A) {
227 auto n = A.rows();
228 auto L = cholesky(A);
229 // Solve L * Y = I (forward substitution)
230 Arr Y(n, n);
231 for (size_t j = 0; j < n; ++j) {
232 for (size_t i = 0; i < n; ++i) {
233 double s = (i == j) ? 1.0 : 0.0;
234 for (size_t k = 0; k < i; ++k) s -= L(i, k) * Y(k, j);
235 Y(i, j) = s / L(i, i);
236 }
237 }
238 // Solve L^T * X = Y (back substitution)
239 Arr X(n, n);
240 for (size_t j = 0; j < n; ++j) {
241 for (size_t i = n; i-- > 0;) {
242 double s = Y(i, j);
243 for (size_t k = i + 1; k < n; ++k) s -= L(k, i) * X(k, j);
244 X(i, j) = s / L(i, i);
245 }
246 }
247 return X;
248}
